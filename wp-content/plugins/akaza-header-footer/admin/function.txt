/**
 * Can validate field by comparing value with other field.
 */
class WPDesk_FCF_Validation_Confirm_Field {
	/**
	 * Field to compare with validated field.
	 *
	 * @var string
	 */
	private $field_to_compare;
	/**
	 * WPDesk_FCF_Validation_Confirm_Field constructor.
	 *
	 * @param string $field_to_compare .
	 */
	public function __construct( $field_to_compare ) {
		$this->field_to_compare = $field_to_compare;
	}
	public function hooks() {
		add_filter( 'flexible_checkout_fields_custom_validation', array( $this, 'register_custom_validation' ) );
	}
	/**
	 * Register custom validation.
	 *
	 * @param array $custom_validation .
	 *
	 * @return array
	 */
	public function register_custom_validation( $custom_validation ) {
		$custom_validation[ 'field_confirmation_' . $this->field_to_compare ] = array(
			'label'    => sprintf( __( 'Compare with %1$s', 'wpdesk' ), $this->field_to_compare ),
			'callback' => array( $this, 'validate' )
		);
		return $custom_validation;
	}
	/**
	 * Validate.
	 *
	 * @param string $field_label Field label.
	 * @param string $value Field value.
	 */
	public function validate( $field_label, $value ) {
		$field_to_compare_value = sanitize_text_field( $_POST[ $this->field_to_compare ] );
		$valid = $field_to_compare_value === $value;
		if ( ! $valid ) {
			wc_add_notice( sprintf( __( 'Invalid %1$s value.', 'wpdesk' ), '<strong>' . $field_label . '</strong>' ), 'error' );
		}
	}
}

// Load amp-web-push script only if <amp-web-push-widget> is used
add_action('amp_post_template_head', function () {
    global $post;

    // Safely get the post content
    $content = isset($post->post_content) ? $post->post_content : '';

    // Check if <amp-web-push-widget> exists in content
    if (strpos($content, '<amp-web-push-widget') !== false) {
        echo '<script async custom-element="amp-web-push" src="https://cdn.ampproject.org/v0/amp-web-push-0.1.js"></script>';
    }
});


// Completely remove amp-web-push script, style, and widget if not used
add_filter('amp_post_template_data', function ($data) {
    ob_start(function ($html) {
        // Remove <script> for amp-web-push
        $html = preg_replace(
            '/<script[^>]+src="https:\/\/cdn\.ampproject\.org\/v0\/amp-web-push-0\.1\.js"[^>]*><\/script>/i',
            '',
            $html
        );

        // Remove <style amp-extension="amp-web-push">
        $html = preg_replace(
            '/<style amp-extension="amp-web-push">.*?<\/style>/is',
            '',
            $html
        );

        // Remove any <amp-web-push-widget> tag if present
        $html = preg_replace(
            '/<amp-web-push-widget[^>]*>.*?<\/amp-web-push-widget>/is',
            '',
            $html
        );

        return $html;
    });
    return $data;
});

// AJAX Live Search for Newsletters
function newsletter_live_search() {
    $term = sanitize_text_field($_GET['term'] ?? '');
    $results = [];

    if ($term) {
        $query = new WP_Query([
            'post_type' => 'post',
            'posts_per_page' => 8,
            's' => $term,
            'category_name' => 'newsletter',
        ]);

        while ($query->have_posts()) {
            $query->the_post();
            $results[] = [
                'title' => get_the_title(),
                'link' => get_permalink(),
            ];
        }
        wp_reset_postdata();
    }

    wp_send_json($results);
}
add_action('wp_ajax_newsletter_search', 'newsletter_live_search');

add_action('rest_api_init', function() {
    register_rest_route('custom-amp/v1', '/search', [
        'methods' => 'GET',
        'callback' => 'custom_amp_search_results',
        'permission_callback' => '__return_true',
    ]);

function custom_amp_search_results($request) {
    $query = sanitize_text_field($request->get_param('s'));
    $args = [
        'post_type' => 'post',
        's' => $query,
        'posts_per_page' => 10,
        'category__not_in' => [ get_cat_ID('newsletter') ]
    ];

    $posts = get_posts($args);
    $results = [];

    foreach ($posts as $post) {
        $results[] = [
            'title' => get_the_title($post->ID),
            'link' => get_permalink($post->ID),
            'excerpt' => wp_trim_words($post->post_excerpt ?: $post->post_content, 20),
            'date' => get_the_date('', $post->ID),
        ];
    }

    return $results;
}

// Force custom single templates for specific categories
function force_category_single_templates($single_template) {
    global $post;

    if (!$post) {
        return $single_template;
    }

    // Newsletter category
    if (is_single() && has_category('newsletter', $post)) {
        $custom_template = get_stylesheet_directory() . '/view_newsletter.php';
        if (file_exists($custom_template)) {
            return $custom_template;
        }
    }

    // Course category
    if (is_single() && has_category('course', $post)) {
        $custom_template = get_stylesheet_directory() . '/course-template.php';
        if (file_exists($custom_template)) {
            return $custom_template;
        }
    }

    return $single_template;
}

add_filter('single_template', 'force_category_single_templates');
